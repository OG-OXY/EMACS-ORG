*** Settings to make mpv an image viewer
:PROPERTIES:
:CUSTOM_ID: settings-to-make-mpv-an-image-viewer
:END:
#+begin_src nix
{ config, pkgs, ... }:

{
  programs.mpv = {
    enable = true;
    
    # 1. High-Quality Rendering & Video/Image Settings
    config = {
      # Use the high-quality profile base
      profile = "high-quality";
      # Force use of a high-performance GPU context (Vulkan preferred, fallback to OpenGL)
      gpu-api = "vulkan";
      vo = "gpu-next";
      
      # Advanced Quality Upscaling
      scale = "ewa_lanczossharp"; # High-end Jinc/Lanczos scaling for extreme clarity
      cscale = "lanczos";         # Chroma upscaling
      dscale = "mitchell";        # Downscaling algorithm
      correct-downscaling = "yes";
      linear-downscaling = "yes";
      sigmoid-upscaling = "yes";
      
      # Performance & Behavior
      hwdec = "auto-safe";        # Hardware decoding for videos in your folders
      keep-open = "yes";          # Don't close the window when an image/video ends
      reset-on-next-file = "video-aspect-override,loop-file,speed,sub-delay,video-pan-x,video-pan-y,video-zoom"; # Reset zoom level when flipping to next image
      
      # Fix image behavior (force mpv to recognize images instead of treating them as 1-second audio)
      image-display-duration = "inf"; 
      loop-file = "inf";
    };

    # 2. Keybindings tailored for image browsing
    bindings = {
      # Navigation (Arrow keys or Mouse Wheel)
      "RIGHT" = "playlist-next";
      "LEFT" = "playlist-prev";
      "SPACE" = "playlist-next";
      "BS" = "playlist-prev"; # Backspace to go back
      "WHEEL_UP" = "playlist-prev";
      "WHEEL_DOWN" = "playlist-next";

      # Zooming and Panning (Holding Alt + Arrow Keys or +/-)
      "=" = "add video-zoom 0.1";
      "-" = "add video-zoom -0.1";
      "Alt+left" = "add video-pan-x 0.05";
      "Alt+right" = "add video-pan-x -0.05";
      "Alt+up" = "add video-pan-y 0.05";
      "Alt+down" = "add video-pan-y -0.05";
      "Ctrl+0" = "set video-zoom 0; set video-pan-x 0; set video-pan-y 0"; # Reset zoom
    };

    # 3. Scripts that make mpv behave like an image app
    scripts = with pkgs.mpvScripts; [
      # Crucial: Automatically grabs all other images/videos in the directory 
      # so you don't have to open them one by one.
      mpris 
      autoload 
    ];
  };

  # Optional: Create a quick terminal alias 'mvi' so you don't confuse it with normal video playback
  home.shellAliases = {
    mvi = "mpv --image-display-duration=inf";
  };
}
#+end_src
